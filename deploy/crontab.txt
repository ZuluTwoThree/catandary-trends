# Catandary Trends — canonical cron configuration (single source of truth).
#
# Install:  edit the three paths below, then  `crontab deploy/crontab.txt`.
# This replaces the former root-level crontab.txt (now a pointer). Paths are
# variables so the SAME file works on the dev box (/home/dirk/...) and a deploy
# host (/opt/...) — change them HERE only, never inline below. No competing
# /opt-vs-/home truths.
CATANDARY_DIR=/home/dirk/projects/catandary-trends
PYTHON=/home/dirk/projects/catandary-trends/.venv/bin/python
LOG_DIR=/home/dirk/logs

# --- core pipeline (active) ---------------------------------------------------
# Full pipeline cycle: poll + LLM + reclassify + auto-publish (every 4 hours).
30 */4 * * *  cd $CATANDARY_DIR && $PYTHON -m pipeline.run_full_cycle --batch 400 >> $LOG_DIR/catandary-cycle.log 2>&1

# Nightly DB backup (02:45; keep 7 days).
45 2 * * *    $PYTHON $CATANDARY_DIR/scripts/backup_db.py --dest /mnt/data-hdd/backups/catandary --skip-sqlite --keep-days 7 >> $LOG_DIR/catandary-backup.log 2>&1

# Weekly source discovery (Sunday 06:00).
0 6 * * 0     $PYTHON $CATANDARY_DIR/scripts/discovery_loop.py >> $LOG_DIR/catandary-discovery-loop.log 2>&1

# Mega-Trend reviewer (1st of month, 03:00).
0 3 1 * *     cd $CATANDARY_DIR && $PYTHON -m pipeline.mega_trend_reviewer >> $LOG_DIR/catandary-mega-reviewer.log 2>&1

# Monthly source-quality check → posts the report to issue #13 (1st of month, 08:00).
0 8 1 * *     cd $CATANDARY_DIR && $PYTHON $CATANDARY_DIR/scripts/monthly_source_check.py --post-issue >> $LOG_DIR/catandary-source-check.log 2>&1

# --- external / opt-in (DISABLED — uncomment to enable) -----------------------

# by default. Uncomment deliberately after the prerequisite is in place.
